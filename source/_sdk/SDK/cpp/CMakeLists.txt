cmake_minimum_required(VERSION 3.16)
project(robobaton_neo_cpp LANGUAGES CXX)
find_package(PkgConfig REQUIRED)
pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl)
pkg_check_modules(JSONC REQUIRED IMPORTED_TARGET json-c)
add_library(robobaton_neo src/client.cpp)
target_compile_features(robobaton_neo PUBLIC cxx_std_17)
target_include_directories(robobaton_neo PUBLIC include)
target_link_libraries(robobaton_neo PUBLIC PkgConfig::CURL PkgConfig::JSONC)
target_compile_options(robobaton_neo PRIVATE -Wall -Wextra -Werror)
add_executable(neo_odometry examples/odometry.cpp)
target_link_libraries(neo_odometry PRIVATE robobaton_neo)
add_executable(neo_image examples/image.cpp)
target_link_libraries(neo_image PRIVATE robobaton_neo)
if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
  enable_testing()
endif()
add_executable(sdk_rbh1_test ../tests/test_rbh1.cpp)
target_link_libraries(sdk_rbh1_test PRIVATE robobaton_neo)
add_test(NAME sdk_rbh1 COMMAND sdk_rbh1_test)
set_tests_properties(sdk_rbh1 PROPERTIES TIMEOUT 15)

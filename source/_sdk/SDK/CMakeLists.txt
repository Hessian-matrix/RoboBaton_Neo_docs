cmake_minimum_required(VERSION 3.16)
project(robobaton_neo_sdk LANGUAGES C CXX)
enable_testing()
add_subdirectory(cpp)
add_executable(sdk_serial_test tests/test_serial.c stm32/neo_mavlink.c)
target_include_directories(sdk_serial_test PRIVATE stm32)
target_compile_features(sdk_serial_test PRIVATE c_std_99)
target_compile_options(sdk_serial_test PRIVATE -Wall -Wextra -Werror)
target_link_libraries(sdk_serial_test PRIVATE m)
add_test(NAME sdk_serial COMMAND sdk_serial_test)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../VIO-Neo/src/io/mavlink_codec.c")
  add_executable(sdk_uart_test tests/test_uart.c stm32/neo_mavlink.c
    stm32/examples/neo_uart.c ../VIO-Neo/src/io/mavlink_codec.c)
  target_include_directories(sdk_uart_test PRIVATE tests/hal_stub stm32 stm32/examples ../VIO-Neo/include)
  target_compile_features(sdk_uart_test PRIVATE c_std_99)
  target_compile_options(sdk_uart_test PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(sdk_uart_test PRIVATE m)
  add_test(NAME sdk_uart_producer COMMAND sdk_uart_test)
  set_tests_properties(sdk_uart_producer PROPERTIES TIMEOUT 15)
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME sdk_python_http_ros
  COMMAND ${CMAKE_COMMAND} -E env SDK_CPP_EXAMPLE=$<TARGET_FILE:neo_odometry> SDK_CPP_IMAGE=$<TARGET_FILE:neo_image>
          ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_sdk.py)
set_tests_properties(sdk_serial sdk_python_http_ros PROPERTIES TIMEOUT 30)
